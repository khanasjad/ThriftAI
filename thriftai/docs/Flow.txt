    if (this.anthropic && products.length > 0) {
      const productData = products.slice(0, 5).map(p => ({
        name: p.name, brand: p.brand, price: p.price,
        originalPrice: p.originalPrice, condition: p.condition, category: p.category
      }))

      const message = await this.anthropic.messages.create({
        model: "claude-3-haiku-20240307",
        max_tokens: 1000,
        temperature: 0.7,
        messages: [{
          role: "user",
          content: `As an expert thrift shopping advisor, analyze this search: "${query}" 
          with budget: $${budget || 'unlimited'}.
          
          I found ${products.length} actual secondhand products in our store:
          ${JSON.stringify(productData, null, 2)}
          
          IMPORTANT: Only discuss the products I've provided above.

          3. Quality Assessment: Evaluation of condition, brand reputation, longevity
          4. Negotiation Tips: Advice for getting best deals on these products

      })

      aiResponse = message.content[0]?.text || ''

  static async searchProductsEnhanced(query: string, budget?: number, sorting?: {}) {
    // Step 5a: Get database configuration
    const [relevantCategories, excludedCategories, keywordMapping] = await Promise.all([
      ConfigurationService.getRelevantCategoriesForIntent(query), // ["ACCESSORIES"]
      ConfigurationService.getExcludedCategoriesForIntent(query), // ["vacuum", "garbage"]
      ConfigurationService.getCategoryKeywordsMapping() // Map<category, keywords[]>
    ])

    // Step 5b: Extract search terms using keyword mapping
    const searchTerms = this.extractSearchTerms(query, keywordMapping)
    // Result: { keywords: ["bags", "designer", "vintage"], categories: ["ACCESSORIES"] }

    // Step 5c: Build intelligent search query
    const where = {
      isAvailable: true,
      AND: [
        { category: { in: ["ACCESSORIES"] } }, // relevant categories
        { category: { notIn: ["VACUUM", "GARBAGE"] } }, // excluded categories  
        budget ? { price: { lte: budget } } : {},
        {
          OR: [
            { name: { contains: "Find vintage designer bags", mode: 'insensitive' } },
            { brand: { contains: "Find vintage designer bags", mode: 'insensitive' } },
            ...buildKeywordSearchConditions(searchTerms, keywordMapping),
            { description: { contains: "Find vintage designer bags", mode: 'insensitive' } },
            { category: { in: ["ACCESSORIES"] } }
          ]
        }
      ]
    }

    // Step 5d: Execute database query
    const products = await prisma.product.findMany({
      where,
      include: { seller: { select: { businessName: true, rating: true } } },
      take: 30,
      orderBy: buildOrderByClause(sorting)
    })

    // Step 5e: Apply intelligent filtering and scoring
    const filteredProducts = await applyIntelligentFiltering(products, query, searchTerms, excludedCategories)
    const scoredProducts = scoreSearchResults(filteredProducts, query, searchTerms)

    return scoredProducts.slice(0, 20) // Return top 20 results
  }
